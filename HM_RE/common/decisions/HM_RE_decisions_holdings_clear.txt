HM_RE_decision_remove_all = {
	picture = {
		reference ="gfx/interface/illustrations/decisions/ep3_decision_city.dds"
	}

    desc = HM_RE_decision_remove_all_desc
	
	selection_tooltip = HM_RE_decision_remove_all_tooltip

	is_shown = {
		global_var:HM_RE_show_decisions ?= yes
		is_landed = yes
		any_directly_owned_province = {
			HM_RE_can_mass_remove_buildings = yes
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		custom_tooltip = HM_RE_remove_all_effect_tooltip
		hidden_effect = {
			every_directly_owned_province = {
				limit = {
					HM_RE_can_mass_remove_buildings = yes
				}
				HM_RE_effect_remove_buildings = yes
			}
		}
	}

	ai_check_interval = 0
}
