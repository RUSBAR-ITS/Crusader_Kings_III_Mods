HM_RE_decision_hide_all = {
	picture = {
		reference ="gfx/interface/illustrations/decisions/ep3_decision_city.dds"
	}
    desc = HM_RE_decision_hide_all_desc
	selection_tooltip = HM_RE_decision_hide_all_tooltip

	is_shown = {
		global_var:HM_RE_show_decisions ?= yes
	}

	effect = {
		custom_tooltip = HM_RE_hide_all_effect_tooltip
		hidden_effect = {
			set_global_variable = {
				name = HM_RE_show_decisions
				value = no
			}
		}
	}

	ai_check_interval = 0
}

HM_RE_decision_show_all = {
	picture = {
		reference ="gfx/interface/illustrations/decisions/ep3_decision_city.dds"
	}
    desc = HM_RE_decision_show_all_desc
	selection_tooltip = HM_RE_decision_show_all_tooltip

	is_shown = {
		OR = {
			NOT = { exists = global_var:HM_RE_show_decisions }
			global_var:HM_RE_show_decisions = no
		}
	}

	effect = {
		custom_tooltip = HM_RE_show_all_effect_tooltip
		hidden_effect = {
			set_global_variable = {
				name = HM_RE_show_decisions
				value = yes
			}
		}
	}

	ai_check_interval = 0
}
