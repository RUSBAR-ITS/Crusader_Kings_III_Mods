HM_RE_decision_buildings_upgrade = {
	picture = {
		reference ="gfx/interface/illustrations/decisions/ep3_decision_city.dds"
	}
    desc = HM_RE_decision_buildings_upgrade_desc
	selection_tooltip = HM_RE_decision_buildings_upgrade_tooltip

	cost = {
		gold = {
			every_directly_owned_province = {
				limit = {
					HM_RE_can_mass_manage_buildings = yes
				}
				add = HM_RE_value_upgrade_buildings_gold_cost
			}
		}
		prestige = {
			every_directly_owned_province = {
				limit = {
					HM_RE_can_mass_manage_buildings = yes
				}
				add = HM_RE_value_upgrade_buildings_prestige_cost
			}
		}
	}

	is_shown = {
		global_var:HM_RE_show_decisions ?= yes
		is_landed = yes
		any_directly_owned_province = {
			count > 0
			HM_RE_can_mass_upgrade_buildings = yes
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		custom_tooltip = HM_RE_effect_upgrade_buildings_tooltip
		show_as_tooltip = {
			every_directly_owned_province = {
				limit = {
					HM_RE_can_mass_manage_buildings = yes
				}
				HM_RE_effect_upgrade_buildings = yes
			}
		}
		hidden_effect = {
			every_directly_owned_province = {
				limit = {
					HM_RE_can_mass_manage_buildings = yes
				}
				HM_RE_effect_upgrade_buildings = yes
			}
		}
	}

	ai_check_interval = 0
}
