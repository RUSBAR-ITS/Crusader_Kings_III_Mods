HM_RE_can_construct_temple_02 = {
	has_building = temple_01

	# Vanilla can_construct_potential
	HM_RE_building_requirement_tribal = no
	province_owner = {
		NOT = { government_has_flag = government_is_mandala }
	}

	# Vanilla can_construct_showing_failures_only
	NOT = { province_owner = { has_character_modifier = tgp_rebuilder_modifier } }

	# Vanilla can_construct
	trigger_if = {
		limit = {
			NOT = {
				province_owner.culture = {
					has_cultural_parameter = temple_next_level
				}
			}
		}
		province_owner.culture = {
			has_innovation = innovation_manorialism
		}
	}
	trigger_else = {
		province_owner.culture = {
			has_cultural_parameter = temple_next_level
			has_innovation = innovation_city_planning
		}
	}
}
