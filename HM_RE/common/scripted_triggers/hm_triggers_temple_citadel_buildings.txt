hm_trigger_construct_temple_citadel_02 = {
	has_building = temple_citadel_01
	hm_building_requirement_tribal = no
	trigger_if = {
		limit = { debug_only = no }
		trigger_if = {
			limit = {
				NOT = {
					county.holder.culture = {
						has_cultural_parameter = temple_next_level
					}
				}
			}
			county.holder.culture = {
				has_innovation = innovation_manorialism
			}
		}
		trigger_else = {
			county.holder.culture = {
				has_cultural_parameter = temple_next_level
				has_innovation = innovation_city_planning
			}
		}
	}
}
