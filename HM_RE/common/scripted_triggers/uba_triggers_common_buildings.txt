hm_trigger_construct_uba_residence_01 = {
	NOT = { has_building_or_higher = uba_residence_01 }
	hm_building_requirement_tribal = no
	NOT = { has_game_rule = uba_common_buildings_off }
	OR = {
		county.holder = {
			any_directly_owned_province = {
				OR = {
					trigger_if = { 
						limit = { has_global_variable = is_uba_loaded }
						OR = {
							has_building_or_higher = UBA_palace_03
							has_building_or_higher = UBA_test_building_1
						}
					}
				}
			}
		}
		trigger_if = { 
			limit = { has_global_variable = is_rba_loaded }
			culture = { has_innovation = innovation_rba_private_estate }
			county.holder = {
				highest_held_title_tier >= tier_duchy
			}
		}
	}
	building_requirement_castle_city_church = { LEVEL = 01 }
}

hm_trigger_construct_uba_kiln_01 = {
	NOT = { has_building_or_higher = uba_kiln_01 }
	hm_building_requirement_tribal = no
	NOT = { has_game_rule = uba_common_buildings_off }
	culture = { has_cultural_parameter = unlock_dragon_kiln_buildings }
	OR = {
		trigger_if = { 
			limit = { has_global_variable = is_rba_loaded }
			OR = {
				terrain = mountains
				terrain = desert_mountains
				terrain = hills
				terrain = terraced_hills
				culture = { has_cultural_parameter = can_build_quarries_everywhere }
				county.holder = {
					highest_held_title_tier >= tier_empire
				}
			}
		}
		county.holder = {
			any_directly_owned_province = {
				OR = {
					trigger_if = { 
						limit = { has_global_variable = is_uba_loaded }
						OR = {
							has_building_or_higher = UBA_workshop_03
							has_building_or_higher = UBA_test_building_1
						}
					}
				}
			}
		}
	}
	building_requirement_castle_city_church = { LEVEL = 01 }
}
