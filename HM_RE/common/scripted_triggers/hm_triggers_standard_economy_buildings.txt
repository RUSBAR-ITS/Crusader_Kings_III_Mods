### watermills

hm_trigger_construct_watermills_01 = {
	NOT = { has_building_or_higher = watermills_01 }
	NOR = {
		has_building_or_higher = caravanserai_01
		has_building_or_higher = windmills_01
	}
	hm_building_requirement_tribal = no
	OR = {
		building_watermills_requirement_terrain = yes
		AND = {
			has_building_or_higher = logging_camps_04
			culture = {
				has_cultural_parameter = watermills_forestries_unlock
			}
		}
	}
	building_requirement_castle_city_church = { LEVEL = 01 }
	trigger_if = {
		limit = { debug_only = no }
		trigger_if = {
			limit = {
				NOT = {
					county.holder.culture = {
						has_cultural_parameter = next_level_w_mills
					}
					culture = {
						has_cultural_parameter = watermills_forestries_unlock
					}
				}
			}
			county.holder.culture = {
				has_innovation = innovation_windmills
			}
		}
		trigger_else = {
			OR = {
				county.holder.culture = {
					has_cultural_parameter = next_level_w_mills
					has_innovation = innovation_manorialism
				}
				AND = {
					has_building_or_higher = logging_camps_04
					culture = {
						has_cultural_parameter = watermills_forestries_unlock
					}
				}
			}
		}
		OR = {
			is_county_capital = yes
			AND = {
				has_holding_type = city_holding
				has_building_or_higher = city_01
				county.holder.culture = { has_cultural_parameter = watermills_windmills_cities }
				NOR = {
					has_building_or_higher = caravanserai_01
					has_building_or_higher = windmills_01
				}
			}
		}
	}
