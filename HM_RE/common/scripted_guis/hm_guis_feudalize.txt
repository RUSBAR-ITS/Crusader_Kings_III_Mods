hm_feudalize_button = {
	scope = character
	saved_scopes = { gui_title }

	is_shown = {
		NOR = {
			government_has_flag = government_is_tribal
			government_has_flag = government_is_nomadic
			government_has_flag = government_is_herder
		}
		trigger_if = {
			limit = {
				has_game_rule = hm_rule_can_feudalize_holdings_vassals
			}
			completely_controls = scope:gui_title
			county.holder = {
				NOT = { government_has_flag = government_is_tribal }
			}
		}
		trigger_else = {
			scope:gui_title = {
				holder = root
			}
		}
		scope:gui_title = {
			is_capital_barony = no # we'll use the county in that case
			title_province = {
				OR = {
					has_holding_type = tribal_holding
					has_holding_type = nomad_holding
					has_holding_type = herder_holding
				}
			}
		}
	}
	
	effect = {
		if = {
			limit = {
				has_game_rule = hm_rule_can_feudalize_holdings_vassals
			}
			open_interaction_window = {
				interaction = hm_feudalize_vassal_holding_interaction
				actor = root
				recipient = root
			}
		}
		else_if = {
			open_interaction_window = {
				interaction = feudalize_holding_interaction
				actor = root
				recipient = root
			}
		}
	}
}
