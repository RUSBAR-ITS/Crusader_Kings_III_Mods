hm_build_all_button = {
    scope = character
	saved_scopes = { gui_holding }
    is_shown = {
		NOT = { has_game_rule = hm_rule_player_extra_slots_disabled }
		has_game_rule = hm_build_all_button_on
		scope:gui_holding = {
			NOT = { has_holding_type = nomad_holding }
			barony = { is_under_holy_order_lease = no }
			OR = {
				hm_trigger_player_add_slot = yes
				has_free_building_slot = yes
			}
		}
		trigger_if = {
			limit = {
				debug_only = no
			}
			trigger_if = {
				limit = {
					scope:gui_holding.barony = { is_leased_out = no }
				}
				OR = {
					scope:gui_holding.province_owner = root
					is_liege_or_above_of = scope:gui_holding.province_owner
				}
				
			}
			trigger_else = {
				OR = {
					scope:gui_holding.barony.lessee = root
					is_liege_or_above_of = scope:gui_holding.barony.lessee
				}
			}
		}
    }
	is_valid = {
		scope:gui_holding = {
			hm_trigger_construct_buildings = yes
			has_ongoing_construction = no
			trigger_if = {
				limit = {
					debug_only = no
					root = {
						has_treasury = yes
					}
					has_game_rule = hm_player_cost_treasury
				}
				hm_value_max_building_slots_and_construct_buildings_gold_cost <= root.treasury
				hm_value_construct_buildings_prestige_cost <= root.prestige
			}
			trigger_else = {
				limit = {
					debug_only = no
				}
				hm_value_max_building_slots_and_construct_buildings_gold_cost <= root.gold
				hm_value_construct_buildings_prestige_cost <= root.prestige
			}
		}
    }
    effect = {
		scope:gui_holding = {
			if = {
				limit = {
					debug_only = no
					root = {
						has_treasury = yes
					}
					has_game_rule = hm_player_cost_treasury
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_max_building_slots_and_buildings_construction_treasury
					value = hm_value_max_building_slots_and_construct_buildings_gold_cost
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_buildings_construction_prestige_cost
					value = {
						value = hm_value_construct_buildings_prestige_cost
						multiply = -1
					}
				}
				root = {
					remove_treasury = scope:hm_gui_scope_max_building_slots_and_buildings_construction_treasury
					add_prestige = scope:hm_gui_scope_buildings_construction_prestige_cost
				}
			}
			else_if = {
				limit = {
					debug_only = no
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_max_building_slots_and_buildings_construction_gold
					value = hm_value_max_building_slots_and_construct_buildings_gold_cost
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_buildings_construction_prestige_cost
					value = {
						value = hm_value_construct_buildings_prestige_cost
						multiply = -1
					}
				}
				root = {
					remove_long_term_gold = scope:hm_gui_scope_max_building_slots_and_buildings_construction_gold
					add_prestige = scope:hm_gui_scope_buildings_construction_prestige_cost
				}
			}
			custom_tooltip = hm_build_all.tt
			hidden_effect = {
				hm_effect_construct_buildings = yes
			}
		}
    }
}

hm_add_slot_button = {
    scope = character
	saved_scopes = { gui_holding }
    is_shown = {
		NOT = { has_game_rule = hm_rule_player_extra_slots_disabled }
		scope:gui_holding = {
			barony = { is_under_holy_order_lease = no }
			NOT = { has_holding_type = nomad_holding }
			hm_trigger_player_add_slot = yes
			hm_value_player_potential_slots > 0
		}
		trigger_if = {
			limit = {
				debug_only = no
			}
			trigger_if = {
				limit = {
					scope:gui_holding.barony = { is_leased_out = no }
				}
				OR = {
					scope:gui_holding.province_owner = root
					is_liege_or_above_of = scope:gui_holding.province_owner
				}
				
			}
			trigger_else = {
				OR = {
					scope:gui_holding.barony.lessee = root
					is_liege_or_above_of = scope:gui_holding.barony.lessee
				}
			}
		}
    }
	is_valid = {
		scope:gui_holding = {
			has_ongoing_construction = no
			trigger_if = {
				limit = {
					debug_only = no
					root = {
						has_treasury = yes
					}
					has_game_rule = hm_player_cost_treasury
				}
				hm_value_player_slot_cost <= root.treasury
			}
			trigger_else = {
				limit = {
					debug_only = no
				}
				hm_value_player_slot_cost <= root.gold
			}
		}
    }
    effect = {
		scope:gui_holding = {
			if = {
				limit = {
					debug_only = no
					root = {
						has_treasury = yes
					}
					has_game_rule = hm_player_cost_treasury
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_add_building_slot_cost
					value = hm_value_player_slot_cost
				}
				root = {
					remove_treasury = scope:hm_gui_scope_add_building_slot_cost
				}
			}
			else_if = {
				limit = {
					debug_only = no
				}
				save_temporary_scope_value_as = {
					name = hm_gui_scope_add_building_slot_cost
					value = hm_value_player_slot_cost
				}
				root = {
					remove_long_term_gold = scope:hm_gui_scope_add_building_slot_cost
				}
			}
			add_province_modifier = extra_building_slot
		}
    }
}
