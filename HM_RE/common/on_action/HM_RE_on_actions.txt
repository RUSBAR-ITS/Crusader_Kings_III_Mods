on_game_start = {
	on_actions = {
		HM_RE_on_action_game_start
	}
}

HM_RE_on_action_game_start = {
	effect = {
		if = {
			limit = {
				NOT = { exists = global_var:HM_RE_is_loaded }
			}
			set_global_variable = {
				name = HM_RE_is_loaded
				value = yes
			}
		}
		if = {
			limit = {
				NOT = { exists = global_var:HM_RE_show_decisions }
			}
			set_global_variable = {
				name = HM_RE_show_decisions
				value = no
			}
		}
	}
}
