on_game_start = {
	on_actions = {
		hm_on_action_game_start
	}
}

hm_on_action_game_start = {
	effect = {
		if = {
			limit = {
				NOT = { exists = global_var:hm_is_loaded }
			}
			set_global_variable = {
				name = hm_is_loaded
				value = yes
			}
		}
		if = {
			limit = {
				NOT = { exists = global_var:hm_show_decisions }
			}
			set_global_variable = {
				name = hm_show_decisions
				value = no
			}
		}
	}
}
