hm_effect_construct_capital_bureau = {
	if = {
		limit = { hm_trigger_construct_capital_bureau = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = capital_bureau_01
		}
		else_if = {
			limit = { hm_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = capital_bureau_01
		}
	}
}
