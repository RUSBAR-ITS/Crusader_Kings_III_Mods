hm_effect_construct_horse = {
	if = {
		limit = { hm_trigger_construct_horse = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = horse_01
		}
		else_if = {
			limit = { hm_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = horse_01
		}
	}
}
