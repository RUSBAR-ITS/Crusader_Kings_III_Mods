hm_effect_construct_dingjudian = {
	if = {
		limit = { hm_trigger_construct_dingjudian = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = dingjudian_01
		}
		else_if = {
			limit = { hm_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = dingjudian_01
		}
	}
}
hm_effect_upgrade_dingjudian = {
	if = {
		limit = { hm_trigger_construct_dingjudian_02 = yes }
		add_building = dingjudian_02
	}
	else_if = {
		limit = { hm_trigger_construct_dingjudian_03 = yes }
		add_building = dingjudian_03
	}
}
hm_effect_remove_dingjudian = {
	if = {
		limit = { has_building = dingjudian_01 }
		remove_building = dingjudian_01
	}
	else_if = {
		limit = { has_building =dingjudian_02 }
		remove_building = dingjudian_02
	}
	else_if = {
		limit = { has_building = dingjudian_03 }
		remove_building = dingjudian_03
	}
}
