HM_RE_effect_upgrade_temple_citadel = {
	if = {
		limit = { HM_RE_can_construct_temple_citadel_02 = yes }
		add_building = temple_citadel_02
	}
	else_if = {
		limit = { HM_RE_can_construct_temple_citadel_03 = yes }
		add_building = temple_citadel_03
	}
	else_if = {
		limit = { HM_RE_can_construct_temple_citadel_04 = yes }
		add_building = temple_citadel_04
	}
}

	if = {
		limit = { HM_RE_can_construct_citadel_shrine = yes }
		if = {
			limit = { has_free_building_slot = yes }

			limit = { HM_RE_trigger_add_slot = yes }
