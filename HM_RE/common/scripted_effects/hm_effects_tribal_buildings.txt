hm_effect_construct_longhouses = {
	if = {
		limit = { hm_trigger_construct_longhouses = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = longhouses_01
		}
		else_if = {
			limit = { hm_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = longhouses_01
		}
	}
