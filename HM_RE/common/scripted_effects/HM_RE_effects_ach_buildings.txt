HM_RE_effect_construct_oath_warriors_building = {
	if = {
		limit = { HM_RE_can_construct_oath_warriors_building = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = oath_warriors_building_01
		}
		else_if = {
			limit = { HM_RE_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = oath_warriors_building_01
		}
	}
}
HM_RE_effect_upgrade_oath_warriors_building = {
	if = {
		limit = { HM_RE_can_construct_oath_warriors_building_02 = yes }
		add_building = oath_warriors_building_02
	}
	else_if = {
		limit = { HM_RE_can_construct_oath_warriors_building_03 = yes }
		add_building = oath_warriors_building_03
	}
	else_if = {
		limit = { HM_RE_can_construct_oath_warriors_building_04 = yes }
		add_building = oath_warriors_building_04
	}
}
HM_RE_effect_remove_oath_warriors_building = {
	if = {
		limit = { has_building = oath_warriors_building_01 }
		remove_building = oath_warriors_building_01
	}
	else_if = {
		limit = { has_building = oath_warriors_building_02 }
		remove_building = oath_warriors_building_02
	}
	else_if = {
		limit = { has_building = oath_warriors_building_03 }
		remove_building = oath_warriors_building_03
	}
	else_if = {
		limit = { has_building = oath_warriors_building_04 }
		remove_building = oath_warriors_building_04
	}
}
