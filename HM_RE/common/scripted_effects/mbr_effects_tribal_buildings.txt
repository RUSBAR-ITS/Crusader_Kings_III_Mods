hm_effect_construct_Townsquare = {
	if = {
		limit = { hm_trigger_construct_Townsquare = yes }
		if = {
			limit = { has_free_building_slot = yes }
			add_building = Townsquare_01
		}
		else_if = {
			limit = { hm_trigger_add_slot = yes }
			add_province_modifier = extra_building_slot
			add_building = Townsquare_01
		}
	}
}
hm_effect_upgrade_Townsquare = {
	if = {
		limit = { hm_trigger_construct_Townsquare_02 = yes }
		add_building = Townsquare_02
	}
	else_if = {
		limit = { hm_trigger_construct_Townsquare_03 = yes }
		add_building = Townsquare_03
	}
}
hm_effect_remove_Townsquare = {
	if = {
		limit = { has_building = Townsquare_01 }
		remove_building = Townsquare_01
	}
	else_if = {
		limit = { has_building =Townsquare_02 }
		remove_building = Townsquare_02
	}
	else_if = {
		limit = { has_building = Townsquare_03 }
		remove_building = Townsquare_03
	}
}
