	}

		value = HM_RE_value_construct_buildings_gold_cost_cap_cheap
	}
	else_if = {

		}

	}
	else_if = {
		limit = {
			has_game_rule = HM_RE_value_construct_buildings_gold_cost_cap_very_expensive
		}
		value = HM_RE_value_construct_buildings_gold_cost_cap_very_expensive
	}
	else = {
		value = HM_RE_value_construct_buildings_gold_cost_cap_default
	}
}

HM_RE_value_construct_buildings_gold_cost_dynamic_cap = {
	value = HM_RE_value_construct_buildings_gold_cost_cap
	
	if = {
		limit = { county.development_level >= 40 }
		multiply = 0.5
	}
	if = {
		limit = { county.development_level >= 30 }
		multiply = 0.7
	}
	if = {
		limit = { county.development_level >= 20 }
		multiply = 0.85
	}
	if = {
		limit = { county.development_level >= 10 }
		multiply = 1.0
	}
	else = {
		multiply = 1.1
	}

	min = 150
	max = 4000
}
