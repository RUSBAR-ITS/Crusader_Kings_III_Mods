hm_value_construct_buildings_gold_cost_dynamic_cap = {
	value = hm_value_construct_buildings_gold_cost_base_cap
	
	if = {
		limit = { county.development_level >= 40 }
		multiply = 0.5
	}
	if = {
		limit = { county.development_level >= 30 }
		multiply = 0.7
	}
	if = {
		limit = { county.development_level >= 20 }
		multiply = 0.85
	}
	if = {
		limit = { county.development_level >= 10 }
		multiply = 1.0
	}
	else = {
		multiply = 1.1
	}

	min = 150
	max = 4000
}
