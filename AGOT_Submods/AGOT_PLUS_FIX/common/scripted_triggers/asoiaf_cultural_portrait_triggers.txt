asoiaf_northern_portrait_trigger = {
	exists = this
	OR = {
		this.culture = culture:first_man
		this.culture = culture:northman
		this.culture = culture:northern_clan #adjust later
		this.culture = culture:barrowman
		this.culture = culture:wolfswood_clansman
		this.culture = culture:bearman
		this.culture = culture:crannogman
		this.culture = culture:krakenman
		this.culture = culture:harborman
		this.culture = culture:sisterman
		this.culture = culture:clawman
		this.culture = {
			any_parent_culture_or_above = {
				OR = {
					this = culture:first_man
					this = culture:northman
					this = culture:northern_clan #adjust later
					this = culture:barrowman
					this = culture:wolfswood_clansman
					this = culture:bearman
					this = culture:crannogman
					this = culture:krakenman
					this = culture:harborman
					this = culture:sisterman
					this = culture:clawman
				}
				NOT = { this = culture:skagosi } #closer to wildlings than northmen
			}
			NOR = {
				has_cultural_pillar = heritage_ironman
				has_cultural_pillar = heritage_wildling
				this = culture:skagosi #closer to wildlings than northmen
			}
		}
	}
	OR = { #all clothing graphics of the main culture(s)
		this.culture = { has_clothing_gfx = first_man_clothing_gfx }
		this.culture = { has_clothing_gfx = northern_clothing_gfx }
		this.culture = { has_clothing_gfx = northman_clothing_gfx }
		this.culture = { has_clothing_gfx = northern_clan_clothing_gfx } #adjust later
		this.culture = { has_clothing_gfx = wolfswood_clansman_clothing_gfx }
		this.culture = { has_clothing_gfx = krakenman_clothing_gfx }
		this.culture = { has_clothing_gfx = harborman_clothing_gfx }
		this.culture = { has_clothing_gfx = sisterman_clothing_gfx }
		this.culture = { has_clothing_gfx = clawman_clothing_gfx }
	}
}
