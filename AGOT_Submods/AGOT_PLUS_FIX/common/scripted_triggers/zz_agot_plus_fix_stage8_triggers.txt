	OR = {
		AND = {
			exists = character:Lowborn_Kings_Landing_3
			this = character:Lowborn_Kings_Landing_3
		}
		has_character_flag = is_Lowborn_Kings_Landing_3
	}
}

	OR = {
		has_inactive_trait = asoiaf_Targaryen_83_trait
		AND = {
			exists = house

		}
	}
}
