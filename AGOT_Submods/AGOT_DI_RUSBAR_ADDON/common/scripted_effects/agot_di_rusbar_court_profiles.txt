# admiral_court_position: Адмирал
agot_di_rusbar_prepare_admiral_court_position_effect = {
	if = { limit = { has_trait = blind } remove_trait = blind }
	if = { limit = { has_trait = infirm } remove_trait = infirm }
	if = { limit = { has_trait = maimed } remove_trait = maimed }
	# Raise base martial to at least 50; engine clamps base skills at zero.
	add_martial_skill = -50
	add_martial_skill = 50
	# Raise base stewardship to at least 50; engine clamps base skills at zero.
	add_stewardship_skill = -50
	add_stewardship_skill = 50
	if = { limit = { has_trait = humble } remove_trait = humble }
	if = { limit = { NOT = { has_trait = authoritative } } add_trait = authoritative }
}
