# Character scope. Applies equally to players, AI rulers and unlanded courtiers.
agot_di_rusbar_process_lifestyle_mastery_effect = {
	if = {
		limit = { agot_di_rusbar_diplomacy_mastered_trigger = yes }
		if = {
			limit = { NOT = { has_character_modifier = agot_di_rusbar_valyrian_diplomat } }
			add_character_modifier = { modifier = agot_di_rusbar_valyrian_diplomat }
		}
		# Native perk-point subtraction clamps at zero; acquired perks are untouched.
		add_diplomacy_lifestyle_perk_points = -1000000
	}
	else_if = {
		limit = { has_character_modifier = agot_di_rusbar_valyrian_diplomat }
		remove_character_modifier = agot_di_rusbar_valyrian_diplomat
	}
