# Character scope. Court wrappers only: Eye and Treatises.
# Reuse the military retraining cleanup without its main-education replacement.
agot_di_rusbar_court_knighthood_effect = {
	if = { limit = { has_trait = education_martial_prowess_1 } remove_trait = education_martial_prowess_1 }
	if = { limit = { has_trait = education_martial_prowess_2 } remove_trait = education_martial_prowess_2 }
	if = { limit = { has_trait = education_martial_prowess_3 } remove_trait = education_martial_prowess_3 }
	if = { limit = { has_trait = education_martial_prowess_4 } remove_trait = education_martial_prowess_4 }
	if = { limit = { NOT = { has_trait = education_martial_prowess_5 } } add_trait = education_martial_prowess_5 }
	agot_di_rusbar_complete_knighthood_effect = yes
}

# Prowess is granted separately. Prowess-only roles use martial education.
# Wet nurse/stooge have no positive skill profile: choose any main education.
# The general skill floor must not influence this selection.
# Keep a matching V education already assigned by the profile (e.g. artificer).

# Roles: court_jester_court_position, court_musician_court_position, cultural_emissary_court_position, cupbearer_court_position, justiciar_court_position, lady_in_waiting_court_position, witness_camp_officer
agot_di_rusbar_court_education_diplomacy_effect = {
	if = {
		limit = {
			NOR = {
				has_trait = education_diplomacy_5
			}
		}
		agot_di_rusbar_clear_court_main_education_effect = yes
		add_trait = education_diplomacy_5
	}
}

		}
	}
