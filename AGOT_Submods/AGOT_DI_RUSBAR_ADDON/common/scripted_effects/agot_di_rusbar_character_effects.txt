	if = { limit = { has_trait = dragon_physique_good_2 } remove_trait = dragon_physique_good_2 }
	if = { limit = { NOT = { has_trait = dragon_physique_good_3 } } add_trait = dragon_physique_good_3 }
	if = { limit = { NOT = { has_trait = dragon_swift } } add_trait = dragon_swift }
	if = { limit = { NOT = { has_trait = dragon_majestic } } add_trait = dragon_majestic }
	# Follow AGOT's hidden fertility convention; do not reveal dragon sex.
	if = {
	    limit = { OR = { is_male = yes NOT = { has_trait = dragon_fertile } } }
	    make_trait_inactive = dragon_fertile
	    if = {
	        limit = { is_female = yes }
	        set_variable = { name = dragon_is_fertile value = yes }
	    }
	}
}
