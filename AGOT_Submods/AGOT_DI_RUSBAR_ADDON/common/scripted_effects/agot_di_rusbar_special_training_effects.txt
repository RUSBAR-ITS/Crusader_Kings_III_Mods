# AGOT 0.5.2.1: squire/knight XP 100 + no ongoing squire story means knight.
# Finish apprenticeship without the full ceremony (prestige, moves, memories, knight tree).
agot_di_rusbar_complete_knighthood_effect = {
	save_scope_as = agot_di_rusbar_retrained_knight
	if = { limit = { has_trait = stripped_knight } remove_trait = stripped_knight }
	if = { limit = { NOT = { has_trait = squire } } add_trait = squire }
	add_trait_xp = { trait = squire track = knight value = 100 }
	# Remove only this character's mentor relation, preserving their own squires.
	every_relation = {
		type = agot_knight
		remove_relation_agot_squire = scope:agot_di_rusbar_retrained_knight
	}
	every_owned_story = {
		limit = { story_type = story_agot_squire_ongoing }
		end_story = yes
	}
	if = { limit = { has_character_flag = agot_knighthood_knightless_squire_flag } remove_character_flag = agot_knighthood_knightless_squire_flag }
	if = { limit = { has_character_flag = agot_favors_being_a_squire_flag } remove_character_flag = agot_favors_being_a_squire_flag }
	if = { limit = { has_character_flag = had_ongoing_squire_story } remove_character_flag = had_ongoing_squire_story }
	if = { limit = { has_character_flag = cannot_be_knighted } remove_character_flag = cannot_be_knighted }
}
