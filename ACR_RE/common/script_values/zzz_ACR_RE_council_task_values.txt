# Additional Campaign Rules. Rusbar Edition
#
# Player-only acceleration for the selected tasks of the five standard council
# positions. Generated task overrides consume these values only after all
# standard penalties and multipliers; targets, side effects and completion
# logic otherwise remain unchanged.

# Percentage/value tasks whose completion scale is 0..100 receive this extra
# monthly progress. Existing progress is preserved, so a naturally faster task
# can still finish earlier than the selected upper bound.
ACR_RE_council_task_progress_boost = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege ?= { is_ai = no }
		}
		if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_year }
			add = {
				value = 100
				divide = 12
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_6_months }
			add = {
				value = 100
				divide = 6
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_3_months }
			add = {
				value = 100
				divide = 3
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_month }
			add = 100
		}
	}
}

# Integrating a title uses the 0..1200 de-jure-drift scale instead of 0..100.
ACR_RE_council_task_integration_boost = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege ?= { is_ai = no }
		}
		if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_year }
			add = 100
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_6_months }
			add = 200
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_3_months }
			add = 400
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_month }
			add = 1200
		}
	}
}

# The standard base values are deliberately not overridden. The generated
# council-task objects add these boosts only after every standard modifier,
# multiplier and minimum-value clamp has finished. This prevents religious,
# cultural, relationship, development and government penalties from reducing
# the selected acceleration.
