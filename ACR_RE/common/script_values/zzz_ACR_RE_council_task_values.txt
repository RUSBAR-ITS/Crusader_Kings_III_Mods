# Additional Campaign Rules. Rusbar Edition
#
# Player-only acceleration for the finite tasks of the five standard council
# positions. We override only the small vanilla progress inputs consumed by
# the current task definitions; the tasks, their targets, side effects and
# completion logic remain entirely standard.

# Percentage/value tasks whose completion scale is 0..100 receive this extra
# monthly progress. Existing progress is preserved, so a naturally faster task
# can still finish earlier than the selected upper bound.
ACR_RE_council_task_progress_boost = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege ?= { is_ai = no }
		}
		if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_year }
			add = {
				value = 100
				divide = 12
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_6_months }
			add = {
				value = 100
				divide = 6
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_3_months }
			add = {
				value = 100
				divide = 3
			}
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_month }
			add = 100
		}
	}
}

# Integrating a title uses the 0..1200 de-jure-drift scale instead of 0..100.
ACR_RE_council_task_integration_boost = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege ?= { is_ai = no }
		}
		if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_year }
			add = 100
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_6_months }
			add = 200
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_3_months }
			add = 400
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_council_task_speed_1_month }
			add = 1200
		}
	}
}

# Chancellor: Integrate Title.
# Standard value: diplomacy * 0.035.
chancellor_integrate_title_progress_gain = {
	add = diplomacy
	multiply = 0.035
	add = ACR_RE_council_task_integration_boost
}
