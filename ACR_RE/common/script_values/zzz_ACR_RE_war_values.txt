# Additional Campaign Rules. Rusbar Edition
#
# This is the current standard truce calculation with one final conditional
# multiplier. Standard deductions and special CB handling remain intact.

standard_truce_duration_days = {
	value = 1825

	# Flexible Truces perk.
	if = {
		limit = {
			scope:attacker = { has_perk = flexible_truces_perk }
		}
		add = -450
	}

	# Certain struggle phases.
	if = {
		limit = {
			scope:attacker = {
				any_character_struggle = {
					is_struggle_parameter_active_secondary_character_involvement_either_trigger = {
						PARAMETER = truces_by_involved_or_interlopers_within_region_shorter
						CHAR = scope:defender
					}
				}
			}
		}
		add = -900
	}
	if = {
		limit = {
			scope:attacker = {
				any_character_struggle = {
					is_struggle_parameter_active_secondary_character_involvement_either_trigger = {
						PARAMETER = truces_by_involved_or_interlopers_within_region_longer
						CHAR = scope:defender
					}
				}
			}
		}
		add = 900
	}

	# Shorter truce if both primary sides are nomads.
	if = {
		limit = {
			scope:attacker = { government_has_flag = government_is_nomadic }
			scope:defender = { government_has_flag = government_is_nomadic }
		}
		subtract = 730
	}

	# Preserve the standard two-year floor before applying the selected rule.
	min = 730

	# Border-raid truces are twice the normally calculated duration.
	if = {
		limit = {
			scope:attacker = {
				any_character_war = {
					primary_attacker = scope:attacker
					primary_defender = scope:defender
					using_cb = fp2_border_raid
				}
			}
		}
		multiply = 2
	}

	# Apply ACR_RE only to the scope selected before campaign start.
	multiply = ACR_RE_truce_duration_multiplier
}

# Root: the war-resolution context used by standard_truce_duration_days.
ACR_RE_truce_duration_multiplier = {
	value = 1

	if = {
		limit = {
			OR = {
				has_game_rule = ACR_RE_truce_scope_all
				AND = {
					has_game_rule = ACR_RE_truce_scope_player_primary
					OR = {
						scope:attacker = { is_ai = no }
						scope:defender = { is_ai = no }
					}
				}
			}
		}

		if = {
			limit = { has_game_rule = ACR_RE_truce_duration_75 }
			multiply = 0.75
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_truce_duration_50 }
			multiply = 0.5
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_truce_duration_25 }
			multiply = 0.25
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_truce_duration_10 }
			multiply = 0.1
		}
		else_if = {
			limit = { has_game_rule = ACR_RE_truce_duration_5 }
			multiply = 0.05
		}
	}
}
