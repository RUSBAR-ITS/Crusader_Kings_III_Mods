namespace = DICM_RBE_ceasar_automation

# Runs Caesar-child automation one day after the sixteenth birthday.
DICM_RBE_ceasar_automation.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_alive = yes
				is_adult = yes
				DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
			}

			DICM_RBE_ceasar_child_adulthood_automation_effect = yes
		}
	}
}

# Evaluates loyalty-curse departure only after title, liege, and court transitions have stabilized.
DICM_RBE_ceasar_automation.0002 = {
	type = character_event
	hidden = yes

	immediate = {
		remove_character_flag = DICM_RBE_loyalty_curse_departure_check_pending

		if = {
			limit = {
				DICM_RBE_loyalty_curse_enabled_trigger = yes
				DICM_RBE_loyalty_curse_immune_trigger = no
				exists = scope:DICM_RBE_loyalty_curse_ceasar
				DICM_RBE_has_left_ceasar_service_trigger = {
					DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_ceasar
				}
			}

			DICM_RBE_loyalty_curse_departure_automation_effect = yes
		}
	}
}

# Evaluates administrative retraining after title and liege transitions have stabilized.
DICM_RBE_ceasar_automation.0003 = {
	type = character_event
	hidden = yes

	immediate = {
		remove_character_flag = DICM_RBE_administrative_retraining_check_pending

		if = {
			limit = {
				DICM_RBE_administrative_retraining_enabled_trigger = yes
				DICM_RBE_administrative_retraining_immune_trigger = no
				has_any_landed_title_trigger = yes
				NOT = {
					has_character_modifier = DICM_RBE_Ceasar_Official
				}
				top_liege = {
					DICM_RBE_is_ceasar_ruler_trigger = yes
				}
			}

			top_liege = {
				save_scope_as = DICM_RBE_administrative_retraining_ceasar
			}
			DICM_RBE_administrative_retraining_automation_effect = {
				DICM_RBE_CULTURE = scope:DICM_RBE_administrative_retraining_ceasar.culture
				DICM_RBE_FAITH = scope:DICM_RBE_administrative_retraining_ceasar.faith
			}
		}
	}
}

# Rolls an eligible wet nurse back from forty-five years and one day to sixteen years and one day.
DICM_RBE_ceasar_automation.0004 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				DICM_RBE_eternal_wet_nurse_age_rollback_trigger = yes
			}

			DICM_RBE_eternal_wet_nurse_age_rollback_effect = yes
		}
	}
}

# Runs after realm settlement or holding construction has finished creating any
# required sub-holding rulers. Vassals already carrying the Official modifier are
# omitted because an earlier individual check has already processed them.
DICM_RBE_ceasar_automation.0005 = {
	type = character_event
	hidden = yes

	immediate = {
		remove_character_flag = DICM_RBE_administrative_retraining_realm_sweep_pending

		if = {
			limit = {
				is_alive = yes
				DICM_RBE_administrative_retraining_enabled_trigger = yes
				DICM_RBE_is_ceasar_ruler_trigger = yes
				has_any_landed_title_trigger = yes
			}

			debug_log = "DICM_RBE: Administrative retraining realm sweep started for [THIS.Char.GetLogName]"
			every_vassal_or_below = {
				limit = {
					is_alive = yes
					has_any_landed_title_trigger = yes
					DICM_RBE_administrative_retraining_immune_trigger = no
					NOT = {
						has_character_modifier = DICM_RBE_Ceasar_Official
					}
				}
				DICM_RBE_schedule_administrative_retraining_check_effect = yes
			}
		}
	}
}
