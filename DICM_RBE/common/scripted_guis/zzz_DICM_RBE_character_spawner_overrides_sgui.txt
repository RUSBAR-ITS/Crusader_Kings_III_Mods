# Late-loading compatibility router for the DICM Chancellor spawn handler.
DI_diplomacy_councillor = {
	scope = character

	effect = {
		if = {
			limit = { DICM_RBE_custom_character_generation_enabled_trigger = yes }
			DICM_RBE_run_custom_councillor_spawner_effect = {
				DICM_RBE_SPAWN_EFFECT = DICM_RBE_spawn_and_assign_chancellor_effect
			}
		}
		else = {
			DICM_RBE_run_original_councillor_spawner_effect = {
				DICM_RBE_TRAIT_EFFECT = DI_diplomacy_traits_effect
				DICM_RBE_COUNCILLOR = councillor_chancellor
			}
		}
	}
}
