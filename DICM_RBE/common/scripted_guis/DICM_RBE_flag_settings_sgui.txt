# Closes the extension flag settings window without changing any setting.
DICM_RBE_close_flag_settings_window = {
	scope = character

	effect = {
		remove_variable = DICM_RBE_flag_settings_window_open
	}
}

# Toggles Caesar-line child inheritance globally.
DICM_RBE_toggle_ceasar_child_inheritance_setting = {
	scope = character

	effect = {
		DICM_RBE_toggle_ceasar_child_inheritance_effect = yes
	}
}

# Reports whether Caesar-line child inheritance is enabled globally.
DICM_RBE_ceasar_child_inheritance_setting_enabled = {
	scope = character

	is_valid = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
	}
}
