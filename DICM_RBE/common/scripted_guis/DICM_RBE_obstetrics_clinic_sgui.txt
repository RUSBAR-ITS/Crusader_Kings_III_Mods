DICM_RBE_close_obstetrics_window = {
	scope = character
	effect = { DICM_RBE_clear_obstetrics_window_state_effect = yes }
}

DICM_RBE_obstetrics_target_is_male = {
	scope = character
	is_valid = {
		trigger_if = {
			limit = { has_variable = DICM_RBE_obstetrics_target }
			var:DICM_RBE_obstetrics_target = { is_male = yes }
		}
		trigger_else = { always = no }
	}
}

DICM_RBE_obstetrics_bastard_pregnancy = {
	scope = character
	is_valid = {
		trigger_if = {
			limit = { has_variable = DICM_RBE_obstetrics_target }
			var:DICM_RBE_obstetrics_target = {
				OR = {
					has_character_flag = bastard_pregnancy
					has_character_flag = unmarried_bastard_pregnancy
				}
			}
		}
		trigger_else = { always = no }
	}
}
