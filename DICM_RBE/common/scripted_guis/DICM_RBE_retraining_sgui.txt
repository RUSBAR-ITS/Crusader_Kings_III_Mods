# Closes the retraining window and clears its transient selection.
DICM_RBE_close_retraining_window = {
	scope = character
	effect = { DICM_RBE_clear_retraining_window_state_effect = yes }
}

# Selects a dynamically listed court-position profile.
DICM_RBE_select_retraining_court_position = {
	scope = character
	saved_scopes = { court_position_type }

	effect = {
		set_variable = {
			name = DICM_RBE_retraining_profile
			value = flag:DICM_RBE_retraining_court_position_profile
		}
		set_variable = {
			name = DICM_RBE_retraining_court_position
			value = scope:court_position_type
		}
	}
}

# Reports whether a dynamically listed court-position profile is selected.
DICM_RBE_retraining_court_position_selected = {
	scope = character
	saved_scopes = { court_position_type }

	is_valid = {
		DICM_RBE_retraining_profile_is_trigger = { PROFILE = DICM_RBE_retraining_court_position_profile }
		trigger_if = {
			limit = { has_variable = DICM_RBE_retraining_court_position }
			var:DICM_RBE_retraining_court_position = scope:court_position_type
		}
		trigger_else = { always = no }
	}
}

# Shows the modifier-only Caesar Child profile only for a living minor target.
DICM_RBE_retraining_minor_ceasar_child_visible = {
	scope = character

	is_valid = {
		trigger_if = {
			limit = { has_variable = DICM_RBE_retraining_target }
			var:DICM_RBE_retraining_target = {
				is_alive = yes
				is_adult = no
			}
		}
		trigger_else = { always = no }
	}
}
