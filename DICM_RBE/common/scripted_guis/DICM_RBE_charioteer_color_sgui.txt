# Closes the charioteer color submenu without spawning a character.
DICM_RBE_close_charioteer_color_window = {
	scope = character

	effect = {
		remove_variable = DICM_RBE_charioteer_color_window_open
	}
}

# Selects the Blue chariot faction.
DICM_RBE_select_charioteer_blue = {
	scope = character

	effect = {
		set_variable = {
			name = DICM_RBE_charioteer_color
			value = flag:DICM_RBE_charioteer_blue
		}
	}
}

# Reports whether the Blue faction is selected.
DICM_RBE_charioteer_blue_selected = {
	scope = character

	is_valid = {
		var:DICM_RBE_charioteer_color = flag:DICM_RBE_charioteer_blue
	}
}

# Confirms the selected faction and starts the regular charioteer spawn batch.
DICM_RBE_confirm_charioteer_spawn = {
	scope = character

	effect = {
		court_position_type:charioteer_court_position = { save_scope_as = court_position_type }
		DICM_RBE_spawn_court_position_batch_effect = yes
		remove_variable = DICM_RBE_charioteer_color_window_open
	}
}
