# Extends the vanilla birth on-action without replacing its existing effect block.
on_birth_child = {
	on_actions = {
		DICM_RBE_on_birth_child_ceasar_inheritance
	}
}

# Grants the child modifier and guaranteed dynastic traits when both biological parents belong to the Caesar line.
DICM_RBE_on_birth_child_ceasar_inheritance = {
	trigger = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
		exists = scope:mother
		exists = scope:real_father
		scope:mother = {
			DICM_RBE_is_ceasar_lineage_parent_trigger = yes
		}
		scope:real_father = {
			DICM_RBE_is_ceasar_lineage_parent_trigger = yes
		}
	}

	effect = {
		if = {
			limit = {
				NOT = {
					has_character_modifier = DICM_RBE_Ceasar_Child
				}
			}
			add_character_modifier = {
				modifier = DICM_RBE_Ceasar_Child
			}
		}

		# Remove birth traits that conflict with the guaranteed Caesar-line traits.
		if = { limit = { has_trait = infertile } remove_trait = infertile }
		if = { limit = { has_trait = inbred } remove_trait = inbred }
		if = { limit = { NOT = { has_trait = fecund } } add_trait = fecund }
		if = { limit = { NOT = { has_trait = pure_blooded } } add_trait = pure_blooded }
		DICM_RBE_genius_mentors_effect = yes
		DICM_RBE_apply_ceasar_family_personality_setting_effect = yes
	}
}
