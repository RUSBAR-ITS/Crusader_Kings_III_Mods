# Applies the complete consort retraining profile when exactly one spouse is a Caesar-line sovereign.
DICM_RBE_on_marriage_assign_ceasar_consort = {
	trigger = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
		exists = scope:spouse
		OR = {
			AND = {
				DICM_RBE_is_ceasar_ruler_trigger = yes
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			AND = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = yes
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				has_character_modifier = DICM_RBE_Ceasar
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			scope:spouse = {
				DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
			}
		}
		else_if = {
			limit = {
				has_character_modifier = DICM_RBE_Ceasar_Female
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			scope:spouse = {
				DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
			}
		}
		else_if = {
			limit = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					has_character_modifier = DICM_RBE_Ceasar
				}
			}
			DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
		}
		else_if = {
			limit = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					has_character_modifier = DICM_RBE_Ceasar_Female
				}
			}
			DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
		}
	}
}

# Promotes the inherited child modifier only when the confirmed player is the previous player's actual heir.
DICM_RBE_on_player_heir_promote_ceasar_child = {
	trigger = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
		exists = scope:previous_player_heir
		this = scope:previous_player_heir
		has_character_modifier = DICM_RBE_Ceasar_Child
	}

	effect = {
		DICM_RBE_promote_ceasar_child_to_ruler_effect = yes
	}
}

# Self-heals missed newly generated landholders without repeatedly retraining
# rulers that already carry the Caesar Official modifier.
DICM_RBE_administrative_retraining_player_realm_maintenance = {
	trigger = {
		is_ai = no
		is_alive = yes
		has_any_landed_title_trigger = yes
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		DICM_RBE_is_ceasar_ruler_trigger = yes
	}

	effect = {
		every_vassal_or_below = {
			limit = {
				is_alive = yes
				has_any_landed_title_trigger = yes
				DICM_RBE_administrative_retraining_immune_trigger = no
				NOT = {
					has_character_modifier = DICM_RBE_Ceasar_Official
				}
			}
			DICM_RBE_schedule_administrative_retraining_check_effect = yes
		}
	}
}

# Schedules a stabilized loyalty-curse check for a character leaving a Caesar sovereign's court.
DICM_RBE_on_character_left_ceasar_court = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		DICM_RBE_loyalty_curse_immune_trigger = no
		exists = scope:old_employer
		scope:old_employer = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
	}

	effect = {
		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:old_employer
		}
	}
}

# Schedules a stabilized check when a realm landholder loses a real landed title.
DICM_RBE_on_loyalty_curse_title_lost = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		DICM_RBE_loyalty_curse_immune_trigger = no
		DICM_RBE_is_ceasar_ruler_trigger = no
		top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
		scope:title = {
			NOR = {
				is_noble_family_title = yes
				is_landless_type_title = yes
				has_variable = adventurer_creation_reason
			}
		}
	}

	effect = {
		top_liege = {
			save_scope_as = DICM_RBE_loyalty_curse_previous_ceasar
		}
		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
		}
	}
}

# Schedules checks for an entire former Caesar vassal branch after its direct liege changes.
DICM_RBE_on_loyalty_curse_vassal_change = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		exists = scope:old_liege
		exists = scope:old_liege.top_liege
		scope:old_liege.top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
	}

	effect = {
		scope:old_liege.top_liege = {
			save_scope_as = DICM_RBE_loyalty_curse_previous_ceasar
		}

		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
		}
		every_vassal_or_below = {
			limit = {
				has_any_landed_title_trigger = yes
			}
			DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
				DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
			}
		}
	}
}

# Schedules retraining whenever a character receives a real landed title. Do not
# require this to be their first title: automatic settlement of a newly completed
# holding may finish assigning several titles before this on-action is evaluated.
# The delayed event checks the final top liege and ignores rulers already prepared
# as Caesar Officials.
DICM_RBE_on_administrative_retraining_title_gain = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		DICM_RBE_administrative_retraining_immune_trigger = no
		scope:title = {
			NOR = {
				is_noble_family_title = yes
				is_landless_type_title = yes
				has_variable = adventurer_creation_reason
			}
		}
	}

	effect = {
		DICM_RBE_schedule_administrative_retraining_check_effect = yes
	}
}

# Handles settlement transitions (tribal, Wanua, nomadic, or herder into a
# settled feudal form). Existing vassals retain their titles during this process,
# therefore each changed vassal is scheduled directly. If the changing character
# is the Caesar sovereign, a delayed realm sweep also catches already-settled
# vassals whose own government did not change.
DICM_RBE_on_administrative_retraining_government_change = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		has_any_landed_title_trigger = yes
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_japan_feudal
		}
		OR = {
			scope:old_government = government_type:tribal_government
			scope:old_government = government_type:wanua_government
			scope:old_government = government_type:nomad_government
			scope:old_government = government_type:herder_government
		}
		OR = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
			AND = {
				DICM_RBE_administrative_retraining_immune_trigger = no
				top_liege = {
					DICM_RBE_is_ceasar_ruler_trigger = yes
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				DICM_RBE_is_ceasar_ruler_trigger = yes
			}
			DICM_RBE_schedule_administrative_retraining_realm_sweep_effect = yes
		}
		else = {
			DICM_RBE_schedule_administrative_retraining_check_effect = yes
		}
	}
}

# root is the province whose construction finished. The county holder is already
# stable even when the holder of the newly created barony is assigned a little later.
DICM_RBE_on_administrative_retraining_building_completed = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		root = {
			exists = county
			county = {
				exists = holder
				holder = {
					exists = top_liege
					top_liege = {
						DICM_RBE_is_ceasar_ruler_trigger = yes
					}
				}
			}
		}
	}

	effect = {
		root.county.holder.top_liege = {
			DICM_RBE_schedule_administrative_retraining_realm_sweep_effect = yes
		}
		debug_log = "DICM_RBE: Building completion requested an administrative retraining realm sweep"
	}
}

# Schedules a delayed check for every baron found or generated by the engine. The
# immediate scope:liege is intentionally not used as a realm filter: during automatic
# settlement of a completed holding the final vassal chain may not yet be stabilized.
DICM_RBE_on_administrative_retraining_baron_created = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		DICM_RBE_administrative_retraining_immune_trigger = no
	}

	effect = {
		DICM_RBE_schedule_administrative_retraining_check_effect = yes
	}
}

# Schedules retraining for an existing landed branch entering a Caesar realm from another realm.
DICM_RBE_on_administrative_retraining_vassal_change = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		has_any_landed_title_trigger = yes
		DICM_RBE_is_ceasar_ruler_trigger = no
		top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
		OR = {
			NOT = {
				exists = scope:old_liege
			}
			scope:old_liege = root
			scope:old_liege.top_liege != root.top_liege
		}
	}

	effect = {
		DICM_RBE_schedule_administrative_retraining_check_effect = yes
		every_vassal_or_below = {
			limit = {
				has_any_landed_title_trigger = yes
			}
			DICM_RBE_schedule_administrative_retraining_check_effect = yes
		}
	}
}
