# Applies the complete consort retraining profile when exactly one spouse is a Caesar-line sovereign.
DICM_RBE_on_marriage_assign_ceasar_consort = {
	trigger = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
		exists = scope:spouse
		OR = {
			AND = {
				DICM_RBE_is_ceasar_ruler_trigger = yes
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			AND = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = yes
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				has_character_modifier = DICM_RBE_Ceasar
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			scope:spouse = {
				DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
			}
		}
		else_if = {
			limit = {
				has_character_modifier = DICM_RBE_Ceasar_Female
				scope:spouse = {
					DICM_RBE_is_ceasar_ruler_trigger = no
				}
			}
			scope:spouse = {
				DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
			}
		}
		else_if = {
			limit = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					has_character_modifier = DICM_RBE_Ceasar
				}
			}
			DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
		}
		else_if = {
			limit = {
				DICM_RBE_is_ceasar_ruler_trigger = no
				scope:spouse = {
					has_character_modifier = DICM_RBE_Ceasar_Female
				}
			}
			DICM_RBE_prepare_true_sovereign_consort_retraining_effect = yes
		}
	}
}

# Promotes the inherited child modifier only when the confirmed player is the previous player's actual heir.
DICM_RBE_on_player_heir_promote_ceasar_child = {
	trigger = {
		DICM_RBE_ceasar_child_inheritance_enabled_trigger = yes
		exists = scope:previous_player_heir
		this = scope:previous_player_heir
		has_character_modifier = DICM_RBE_Ceasar_Child
	}

	effect = {
		DICM_RBE_promote_ceasar_child_to_ruler_effect = yes
	}
}

# Schedules a stabilized loyalty-curse check for a character leaving a Caesar sovereign's court.
DICM_RBE_on_character_left_ceasar_court = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		DICM_RBE_loyalty_curse_immune_trigger = no
		exists = scope:old_employer
		scope:old_employer = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
	}

	effect = {
		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:old_employer
		}
	}
}

# Schedules a stabilized check when a realm landholder loses a real landed title.
DICM_RBE_on_loyalty_curse_title_lost = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		DICM_RBE_loyalty_curse_immune_trigger = no
		DICM_RBE_is_ceasar_ruler_trigger = no
		top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
		scope:title = {
			NOR = {
				is_noble_family_title = yes
				is_landless_type_title = yes
				has_variable = adventurer_creation_reason
			}
		}
	}

	effect = {
		top_liege = {
			save_scope_as = DICM_RBE_loyalty_curse_previous_ceasar
		}
		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
		}
	}
}

# Schedules checks for an entire former Caesar vassal branch after its direct liege changes.
DICM_RBE_on_loyalty_curse_vassal_change = {
	trigger = {
		DICM_RBE_loyalty_curse_enabled_trigger = yes
		exists = scope:old_liege
		scope:old_liege = {
			top_liege = {
				DICM_RBE_is_ceasar_ruler_trigger = yes
			}
		}
	}

	effect = {
		scope:old_liege = {
			top_liege = {
				save_scope_as = DICM_RBE_loyalty_curse_previous_ceasar
			}
		}

		DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
		}
		every_vassal_or_below = {
			limit = {
				has_any_landed_title_trigger = yes
			}
			DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
				DICM_RBE_CEASAR = scope:DICM_RBE_loyalty_curse_previous_ceasar
			}
		}
	}
}

# Schedules retraining when a previously landless character receives their first real landed title in a Caesar realm.
DICM_RBE_on_administrative_retraining_title_gain = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		DICM_RBE_administrative_retraining_immune_trigger = no
		top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
		scope:title = {
			NOR = {
				is_noble_family_title = yes
				is_landless_type_title = yes
				has_variable = adventurer_creation_reason
			}
		}
		NOT = {
			any_held_title = {
				this != scope:title
				NOR = {
					is_noble_family_title = yes
					is_landless_type_title = yes
					has_variable = adventurer_creation_reason
				}
			}
		}
	}

	effect = {
		top_liege = {
			save_scope_as = DICM_RBE_administrative_retraining_current_ceasar
		}
		DICM_RBE_schedule_administrative_retraining_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_administrative_retraining_current_ceasar
		}
	}
}

# Schedules retraining for an existing landed branch entering a Caesar realm from another realm.
DICM_RBE_on_administrative_retraining_vassal_change = {
	trigger = {
		DICM_RBE_administrative_retraining_enabled_trigger = yes
		has_any_landed_title_trigger = yes
		DICM_RBE_is_ceasar_ruler_trigger = no
		top_liege = {
			DICM_RBE_is_ceasar_ruler_trigger = yes
		}
		OR = {
			NOT = {
				exists = scope:old_liege
			}
			scope:old_liege = root
			scope:old_liege.top_liege != root.top_liege
		}
	}

	effect = {
		top_liege = {
			save_scope_as = DICM_RBE_administrative_retraining_current_ceasar
		}

		DICM_RBE_schedule_administrative_retraining_check_effect = {
			DICM_RBE_CEASAR = scope:DICM_RBE_administrative_retraining_current_ceasar
		}
		every_vassal_or_below = {
			limit = {
				has_any_landed_title_trigger = yes
			}
			DICM_RBE_schedule_administrative_retraining_check_effect = {
				DICM_RBE_CEASAR = scope:DICM_RBE_administrative_retraining_current_ceasar
			}
		}
	}
}
