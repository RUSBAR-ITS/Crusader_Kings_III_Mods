# Provides extension-specific genetic modification actions inside the Divine Intervention interaction category.
DICM_RBE_genetic_modification_interaction = {
	interface_priority = 250
	category = interaction_category_cheat_menu
	desc = DICM_RBE_genetic_modification_interaction_desc
	common_interaction = no

	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	send_options_exclusive = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			any_owned_story = {
				story_type = DI_misc_variables_story
				has_variable = DI_cheat_menu
			}
		}
	}

	send_option = {
		flag = DICM_RBE_apply_genetic_modification
		localization = DICM_RBE_genetic_modification_option
	}
	send_option = {
		flag = DICM_RBE_apply_genetic_correction
		localization = DICM_RBE_genetic_correction_option
	}

	on_accept = {
		if = {
			limit = {
				scope:DICM_RBE_apply_genetic_modification = yes
			}

			scope:recipient = {
				DICM_RBE_genetic_modification_effect = yes
			}
		}
		if = {
			limit = {
				scope:DICM_RBE_apply_genetic_correction = yes
			}

			scope:recipient = {
				DICM_RBE_genetic_correction_effect = yes
			}
		}
	}
}
