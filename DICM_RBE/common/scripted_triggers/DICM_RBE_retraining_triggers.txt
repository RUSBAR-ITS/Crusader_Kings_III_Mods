# Returns true when the current retraining selection can be applied to a living target.
DICM_RBE_retraining_selection_ready_trigger = {
	has_variable = DICM_RBE_retraining_target
	has_variable = DICM_RBE_retraining_profile
	var:DICM_RBE_retraining_target = { is_alive = yes }
	OR = {
		AND = {
			var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_court_position_profile
			has_variable = DICM_RBE_retraining_court_position
		}
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_charioteer_blue_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_charioteer_green_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_charioteer_white_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_charioteer_red_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_chancellor_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_marshal_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_steward_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_spymaster_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_court_chaplain_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_knight_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_commander_profile
		var:DICM_RBE_retraining_profile = flag:DICM_RBE_retraining_official_profile
	}
}
