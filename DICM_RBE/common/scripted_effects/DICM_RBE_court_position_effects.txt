# Ensures the wet nurse has the vanilla motherhood aptitude bonus without adding a living dependent.
DICM_RBE_ensure_wet_nurse_has_child_effect = {
	if = {
		limit = {
			NOT = { any_child = { } }
		}
		hidden_effect = {
			save_scope_as = DICM_RBE_wet_nurse
			create_character = {
				age = 0
				gender_female_chance = 50
				location = root.location
				culture = root.culture
				faith = root.faith
				dynasty = none
				trait = sickly
				after_creation = {
					set_mother = scope:DICM_RBE_wet_nurse
				}
				save_scope_as = DICM_RBE_wet_nurse_child
			}
			scope:DICM_RBE_wet_nurse_child = {
				death = { death_reason = death_sickly }
			}
		}
	}
}

# Raise influence to the threshold for the highest vanilla influence level.
DICM_RBE_raise_influence_to_maximum_level_effect = {
	if = {
		limit = { influence < 16000 }
		change_influence = {
			value = 16000
			subtract = influence
		}
	}
}

# Court position preparation effects are assembled in independent layers.
# Each effect applies general bad-trait removal, aptitude-positive personality traits,
# soft positive traits, and removal of aptitude-negative personality traits, in that order.
# Universal Loyal and global sexuality rules are applied by the spawn finalizer.
# Position-specific modifiers are added only where aptitude directly requires them.

# Akolouthos
DICM_RBE_prepare_akolouthos_court_position_effect = {
	DICM_RBE_add_ceasar_squad_modifier_effect = yes
	if = {
		limit = {
			NOR = {
				culture = {
					OR = {
						has_cultural_tradition = tradition_fp1_coastal_warriors
						has_cultural_tradition = tradition_hird
						has_cultural_tradition = tradition_fp1_performative_honour
						has_cultural_pillar = heritage_north_germanic
						has_cultural_pillar = heritage_west_germanic
					}
				}
				culture = {
					NOR = {
						has_same_culture_heritage = scope:target.culture
						has_same_culture_language = scope:target.culture
					}
				}
			}
		}
		set_culture = culture:norse
	}
	DICM_RBE_retrain_martial_education_effect = yes
	DICM_RBE_add_chivalry_tree_perks_effect = yes
	DICM_RBE_add_unyielding_defender_commander_trait_max_xp_effect = yes
	DICM_RBE_add_physique_good_3_aptitude_trait_effect = yes
	DICM_RBE_add_strong_aptitude_trait_effect = yes
	DICM_RBE_add_court_position_trait_if_missing_effect = { TRAIT = shieldmaiden }
	DICM_RBE_add_court_position_trait_if_missing_effect = { TRAIT = varangian }
	DICM_RBE_remove_bad_personality_traits_effect = yes
	DICM_RBE_add_personality_trait_with_one_opposite_effect = { TRAIT = brave OPPOSITE = craven }
	DICM_RBE_add_personality_trait_with_one_opposite_effect = { TRAIT = honest OPPOSITE = deceitful }
	DICM_RBE_add_personality_trait_with_one_opposite_effect = { TRAIT = just OPPOSITE = arbitrary }
	DICM_RBE_add_personality_trait_with_two_opposites_effect = { TRAIT = stubborn OPPOSITE_1 = eccentric OPPOSITE_2 = fickle }
	DICM_RBE_add_good_personality_traits_soft_effect = yes
	DICM_RBE_remove_personality_trait_if_present_effect = { TRAIT = craven }
	DICM_RBE_remove_personality_trait_if_present_effect = { TRAIT = fickle }
}
