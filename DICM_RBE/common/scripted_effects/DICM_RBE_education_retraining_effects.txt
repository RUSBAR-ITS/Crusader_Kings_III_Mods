# Adds Confucian education and raises it to exactly 100 XP.
DICM_RBE_retrain_confucian_education_effect = {
	if = { limit = { NOT = { has_trait = confucian_education } } add_trait = confucian_education }
	save_scope_as = DICM_RBE_confucian_education_target
	if = {
		limit = {
			has_trait = confucian_education
			has_trait_xp = { trait = confucian_education value < 100 }
		}
		add_trait_xp = {
			trait = confucian_education
			value = {
				value = 100
				subtract = "scope:DICM_RBE_confucian_education_target.has_trait_xp(confucian_education)"
			}
		}
	}
}
