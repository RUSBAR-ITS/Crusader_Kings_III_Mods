# Enables Sodom and Gomorrah and disables its mutually exclusive setting.
DICM_RBE_enable_sodom_and_gomorrah_effect = {
	DICM_RBE_disable_chemical_sterilization_effect = yes
	if = {
		limit = {
			NOT = {
				has_global_variable = DICM_RBE_sodom_and_gomorrah_enabled
			}
		}
		set_global_variable = {
			name = DICM_RBE_sodom_and_gomorrah_enabled
			value = yes
		}
		debug_log = "DICM_RBE: DICM_RBE_sodom_and_gomorrah_enabled = yes"
	}
}

# Disables Sodom and Gomorrah without changing characters processed earlier.
DICM_RBE_disable_sodom_and_gomorrah_effect = {
	if = {
		limit = {
			has_global_variable = DICM_RBE_sodom_and_gomorrah_enabled
		}
		remove_global_variable = DICM_RBE_sodom_and_gomorrah_enabled
		debug_log = "DICM_RBE: DICM_RBE_sodom_and_gomorrah_enabled = no"
	}
}

# Toggles Sodom and Gomorrah globally.
DICM_RBE_toggle_sodom_and_gomorrah_effect = {
	if = {
		limit = {
			has_global_variable = DICM_RBE_sodom_and_gomorrah_enabled
		}
		DICM_RBE_disable_sodom_and_gomorrah_effect = yes
	}
	else = {
		DICM_RBE_enable_sodom_and_gomorrah_effect = yes
	}
}
