# Adds one requested non-personality aptitude trait when it is missing.
DICM_RBE_add_court_position_trait_if_missing_effect = {
	if = {
		limit = { NOT = { has_trait = $TRAIT$ } }
		add_trait = $TRAIT$
	}
}

# Replaces the follower status with the Adventurer trait when necessary.
DICM_RBE_add_adventurer_aptitude_trait_effect = {
	if = { limit = { has_trait = adventurer_follower } remove_trait = adventurer_follower }
	DICM_RBE_add_court_position_trait_if_missing_effect = { TRAIT = adventurer }
}

# Replaces the incompatible eunuch trait and adds Beardless Eunuch.
DICM_RBE_add_beardless_eunuch_aptitude_trait_effect = {

	DICM_RBE_add_court_position_trait_if_missing_effect = { TRAIT = beardless_eunuch }
}

# Adds Devoted only when the character's culture grants its aptitude bonus.
DICM_RBE_add_devoted_aptitude_trait_effect = {
	if = {
		limit = {
			culture = { has_cultural_parameter = devoted_trait_bonuses }
			NOT = { has_trait = devoted }
		}
		add_trait = devoted
	}
}

# Adds Varangian to a Bodyguard only when the vanilla conditional bonus can apply.
DICM_RBE_add_bodyguard_varangian_aptitude_trait_effect = {
	if = {
		limit = {
			employer ?= {
				has_title = title:e_byzantium
				employs_court_position = akolouthos_court_position
			}
			NOT = { has_trait = varangian }
		}
		add_trait = varangian
	}
}

# Adds Blademaster and raises its XP to exactly 100.
DICM_RBE_add_lifestyle_blademaster_max_xp_effect = {
	DICM_RBE_add_court_position_trait_if_missing_effect = { TRAIT = lifestyle_blademaster }
	save_scope_as = DICM_RBE_court_position_trait_target
	if = {
		limit = { has_trait_xp = { trait = lifestyle_blademaster value < 100 } }
		add_trait_xp = { trait = lifestyle_blademaster value = { value = 100 subtract = "scope:DICM_RBE_court_position_trait_target.has_trait_xp(lifestyle_blademaster)" } }
	}
}

# Replaces any charioteer faction with the requested faction and sets its XP to exactly 100.
DICM_RBE_set_charioteer_faction_max_xp_effect = {
	if = { limit = { has_trait = charioteer_blue } remove_trait = charioteer_blue }
	if = { limit = { has_trait = charioteer_green } remove_trait = charioteer_green }
	if = { limit = { has_trait = charioteer_white } remove_trait = charioteer_white }
	if = { limit = { has_trait = charioteer_red } remove_trait = charioteer_red }
	add_trait = $TRAIT$
	add_trait_xp = { trait = $TRAIT$ value = 100 }
}

# Creates a maximally experienced Blue charioteer.
DICM_RBE_set_charioteer_blue_max_xp_effect = {
	DICM_RBE_set_charioteer_faction_max_xp_effect = { TRAIT = charioteer_blue }
}
