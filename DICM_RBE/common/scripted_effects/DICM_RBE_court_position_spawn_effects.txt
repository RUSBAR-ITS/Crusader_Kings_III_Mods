# Creates a court-position candidate with deterministic base skills and no random traits.
DICM_RBE_spawn_court_position_character_effect = {
	scope:DI_story = {
		save_scope_value_as = {
			name = DICM_RBE_spawn_age_minimum
			value = var:DI_char_spawner_age_increment_value
		}
		save_scope_value_as = {
			name = DICM_RBE_spawn_female_chance
			value = var:DI_female_chance
		}
	}

	save_scope_value_as = {
		name = DICM_RBE_spawn_learning
		value = 30
	}
	if = {
		limit = { DICM_RBE_court_position_uses_languages_trigger = yes }
		save_scope_value_as = {
			name = DICM_RBE_spawn_learning
			value = 60
		}
	}

	DI_char_spawner_resolve_origin_effect = yes

	create_character = {
		age = DICM_RBE_spawn_age_minimum_value
		gender_female_chance = scope:DICM_RBE_spawn_female_chance
		health = 5
		fertility = 0.5
		employer = scope:target
		faith = scope:DI_cs_spawn_faith
		culture = scope:DI_cs_spawn_culture
		dynasty = none
		diplomacy = 30
		martial = 30
		stewardship = 30
		intrigue = 30
		learning = scope:DICM_RBE_spawn_learning
		prowess = 30
		random_traits = no
		save_scope_as = new_courtier
	}

	# Preserve the independent immortality option while ignoring the stat and trait profile checkboxes.
	scope:DI_story = {
		if = {
			limit = { has_variable = DI_char_spawner_immortal }
			scope:new_courtier = {
				if = {
					limit = { NOT = { has_trait = DI_immortal } }
					add_trait = DI_immortal
				}
			}
		}
	}
}

# Appoints the newly prepared character, replacing an existing holder when the position is full.
DICM_RBE_appoint_spawned_court_position_character_effect = {
	if = {
		limit = { NOT = { DI_can_employ_selected_court_position = yes } }
		DI_revoke_selected_court_position_holder_effect = yes
	}
	appoint_court_position = {
		recipient = scope:new_courtier
		court_position = scope:court_position_type
	}
}

# Complete replacement path for positions covered by DICM RBE profiles.
DICM_RBE_spawn_selected_court_position_effect = {
	DICM_RBE_spawn_court_position_character_effect = yes
	scope:new_courtier = {
		DICM_RBE_prepare_spawned_court_position_character_effect = yes
	}
	DICM_RBE_appoint_spawned_court_position_character_effect = yes
}

# Preserves the original DICM path for intentionally unsupported positions and future additions.
DICM_RBE_spawn_selected_court_position_original_effect = {
	DI_spawn_low_court_character_effect = yes
	scope:new_courtier = {
		DI_apply_court_position_validity_effect = { COURT_POSITION = scope:court_position_type }
		DI_apply_court_position_trait_xp_effect = { COURT_POSITION = scope:court_position_type }
		DI_court_position_post_spawn_effect = { COURT_POSITION = scope:court_position_type }
	}
	DICM_RBE_appoint_spawned_court_position_character_effect = yes
}

# Routes the DICM court-position window to a complete DICM RBE profile or the preserved original path.
DICM_RBE_spawn_selected_court_position_router_effect = {
	if = {
		limit = { DICM_RBE_court_position_uses_custom_spawn_trigger = yes }
		DICM_RBE_spawn_selected_court_position_effect = yes
	}
	else = {
		DICM_RBE_spawn_selected_court_position_original_effect = yes
	}
}

# Runs the unchanged DICM multi-character court-position loop through the DICM RBE router.
DICM_RBE_spawn_court_position_batch_effect = {
	save_scope_as = root_target
	random_owned_story = {
		limit = { story_type = DI_misc_variables_story }
		save_scope_as = DI_story
	}
	scope:DI_story = {
		save_scope_value_as = {
			name = DI_char_spawner_count_increment_value
			value = var:DI_char_spawner_count_increment_value
		}
	}
	DI_apply_court_position_profile_effect = { COURT_POSITION = scope:court_position_type }

	if = {
		limit = { scope:DI_story = { has_variable = DI_pinned_char_var } }
		every_in_list = {
			variable = pinned_characters
			save_scope_as = target
			while = {
				count = scope:DI_char_spawner_count_increment_value
				DICM_RBE_spawn_selected_court_position_router_effect = yes
			}
		}
	}
	else = {
		save_scope_as = target
		while = {
			count = scope:DI_char_spawner_count_increment_value
			DICM_RBE_spawn_selected_court_position_router_effect = yes
		}
	}

	if = {
		limit = { scope:root_target.var:DI_cs_random_traits ?= no }
		DI_no_random_traits_effect = yes
	}
	else_if = {
		limit = { scope:root_target.var:DI_cs_random_traits ?= yes }
		DI_random_traits_effect = yes
	}
}
