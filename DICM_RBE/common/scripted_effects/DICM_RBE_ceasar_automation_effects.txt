# Enables loyalty-curse automation for future departures from a Caesar sovereign's court.
DICM_RBE_enable_loyalty_curse_effect = {
	if = {
		limit = {
			NOT = {
				has_global_variable = DICM_RBE_loyalty_curse_enabled
			}
		}

		set_global_variable = {
			name = DICM_RBE_loyalty_curse_enabled
			value = yes
		}
	debug_log = "DICM_RBE: DICM_RBE_loyalty_curse_enabled = yes"
	}
}

# Disables loyalty-curse automation without changing characters processed earlier.
DICM_RBE_disable_loyalty_curse_effect = {
	if = {
		limit = {
			has_global_variable = DICM_RBE_loyalty_curse_enabled
		}
		remove_global_variable = DICM_RBE_loyalty_curse_enabled
	debug_log = "DICM_RBE: DICM_RBE_loyalty_curse_enabled = no"
	}
}

# Entry point intended for the future cheat-menu setting.
DICM_RBE_toggle_loyalty_curse_effect = {
	if = {
		limit = {
			has_global_variable = DICM_RBE_loyalty_curse_enabled
		}
		DICM_RBE_disable_loyalty_curse_effect = yes
	}
	else = {
		DICM_RBE_enable_loyalty_curse_effect = yes
	}
}

# Schedules one stabilized loyalty-curse departure check for a non-immune character.
DICM_RBE_schedule_loyalty_curse_departure_check_effect = {
	if = {
		limit = {
			DICM_RBE_loyalty_curse_enabled_trigger = yes
			DICM_RBE_loyalty_curse_immune_trigger = no
			NOT = {
				has_character_flag = DICM_RBE_loyalty_curse_departure_check_pending
			}
		}

		add_character_flag = {
			flag = DICM_RBE_loyalty_curse_departure_check_pending
			days = 3
		}
		$DICM_RBE_CEASAR$ = {
			save_scope_as = DICM_RBE_loyalty_curse_ceasar
		}
		trigger_event = {
			id = DICM_RBE_ceasar_automation.0002
			days = 1
		}
	}
}

# Schedules one stabilized administrative-retraining check for a non-child character.
DICM_RBE_schedule_administrative_retraining_check_effect = {
	if = {
		limit = {
			DICM_RBE_administrative_retraining_enabled_trigger = yes
			NOT = {
				has_character_modifier = DICM_RBE_Ceasar_Child
			}
			NOT = {
				has_character_flag = DICM_RBE_administrative_retraining_check_pending
			}
		}

		add_character_flag = {
			flag = DICM_RBE_administrative_retraining_check_pending
			days = 3
		}
		$DICM_RBE_CEASAR$ = {
			save_scope_as = DICM_RBE_administrative_retraining_ceasar
		}
		trigger_event = {
			id = DICM_RBE_ceasar_automation.0003
			days = 1
		}
	}
}

# Applies the complete loyalty curse while preserving The Wake and lifestyle-mastery modifiers.
DICM_RBE_apply_loyalty_curse_effect = {
	if = {
		limit = {
			DICM_RBE_loyalty_curse_immune_trigger = no
		}

		DICM_RBE_adjust_all_base_skills_effect = {
			DICM_RBE_DECREASE_DELTA = -500
			DICM_RBE_INCREASE_DELTA = 0
		}
		DICM_RBE_genetic_curse_effect = yes
		DICM_RBE_add_bad_personality_traits_effect = yes
		DICM_RBE_remove_all_perk_lifestyle_traits_effect = yes
		DICM_RBE_remove_all_event_and_system_lifestyle_traits_effect = yes
		DICM_RBE_remove_all_commander_traits_effect = yes
		DICM_RBE_retrain_learning_education_level_one_effect = yes

		if = { limit = { has_character_modifier = DICM_RBE_Ceasar_Squad } remove_character_modifier = DICM_RBE_Ceasar_Squad }
		if = { limit = { has_character_modifier = DICM_RBE_Ceasar_Master } remove_character_modifier = DICM_RBE_Ceasar_Master }
		if = { limit = { has_character_modifier = DICM_RBE_Ceasar_Official } remove_character_modifier = DICM_RBE_Ceasar_Official }
		if = { limit = { has_character_modifier = DICM_RBE_Ceasar_Wet_Nurse } remove_character_modifier = DICM_RBE_Ceasar_Wet_Nurse }
		if = { limit = { has_character_modifier = DICM_RBE_Ceasar_Scout } remove_character_modifier = DICM_RBE_Ceasar_Scout }
		if = { limit = { NOT = { has_character_modifier = DICM_RBE_Ideal_Stooge } } add_character_modifier = DICM_RBE_Ideal_Stooge }

		debug_log = "DICM_RBE: Loyalty curse applied to a departed character"
	}
}

# Entry point invoked after a character's departure state has stabilized.
DICM_RBE_loyalty_curse_departure_automation_effect = {
	DICM_RBE_apply_loyalty_curse_effect = yes
}
