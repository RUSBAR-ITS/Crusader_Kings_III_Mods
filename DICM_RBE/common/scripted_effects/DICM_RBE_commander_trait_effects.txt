# Adds every compatible standard commander trait at zero XP without changing XP on existing traits.
# Reckless replaces its mutually exclusive Cautious Leader trait.

DICM_RBE_add_all_commander_traits_zero_xp_effect = {
	if = { limit = { NOT = { has_trait = logistician } } add_trait = logistician }
	if = { limit = { NOT = { has_trait = military_engineer } } add_trait = military_engineer }
	if = { limit = { NOT = { has_trait = aggressive_attacker } } add_trait = aggressive_attacker }
	if = { limit = { NOT = { has_trait = unyielding_defender } } add_trait = unyielding_defender }
	if = { limit = { NOT = { has_trait = forder } } add_trait = forder }
	if = { limit = { NOT = { has_trait = flexible_leader } } add_trait = flexible_leader }
	if = { limit = { NOT = { has_trait = desert_warrior } } add_trait = desert_warrior }
	if = { limit = { NOT = { has_trait = jungle_stalker } } add_trait = jungle_stalker }
	if = { limit = { NOT = { has_trait = reaver } } add_trait = reaver }
	if = { limit = { has_trait = cautious_leader } remove_trait = cautious_leader }
	if = { limit = { NOT = { has_trait = reckless } } add_trait = reckless }
	if = { limit = { NOT = { has_trait = holy_warrior } } add_trait = holy_warrior }
	if = { limit = { NOT = { has_trait = open_terrain_expert } } add_trait = open_terrain_expert }
	if = { limit = { NOT = { has_trait = rough_terrain_expert } } add_trait = rough_terrain_expert }
	if = { limit = { NOT = { has_trait = forest_fighter } } add_trait = forest_fighter }
	if = { limit = { NOT = { has_trait = organizer } } add_trait = organizer }
	if = { limit = { NOT = { has_trait = winter_soldier } } add_trait = winter_soldier }
}

# Adds one requested commander trait without changing its XP when already present.
DICM_RBE_add_commander_trait_if_missing_effect = {
	if = {
		limit = { NOT = { has_trait = $TRAIT$ } }
		add_trait = $TRAIT$
	}
}

# Adds Logistician and raises its XP to exactly 100.
DICM_RBE_add_logistician_commander_trait_max_xp_effect = {
	DICM_RBE_add_commander_trait_if_missing_effect = { TRAIT = logistician }
	save_scope_as = DICM_RBE_commander_trait_target
	if = {
		limit = { has_trait_xp = { trait = logistician value < 100 } }
		add_trait_xp = { trait = logistician value = { value = 100 subtract = "scope:DICM_RBE_commander_trait_target.has_trait_xp(logistician)" } }
	}
}
