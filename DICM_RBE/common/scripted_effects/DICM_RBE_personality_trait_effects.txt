# Adds positive personality traits only when neither the trait nor any of its opposites is present.
DICM_RBE_add_good_personality_traits_soft_effect = {
	if = { limit = { NOR = { has_trait = temperate has_trait = gluttonous } } add_trait = temperate }
	if = { limit = { NOR = { has_trait = generous has_trait = greedy } } add_trait = generous }
	if = { limit = { NOR = { has_trait = diligent has_trait = lazy } } add_trait = diligent }
	if = { limit = { NOR = { has_trait = calm has_trait = wrathful } } add_trait = calm }
	if = { limit = { NOR = { has_trait = patient has_trait = impatient } } add_trait = patient }
	if = { limit = { NOR = { has_trait = humble has_trait = arrogant } } add_trait = humble }
	if = { limit = { NOR = { has_trait = honest has_trait = deceitful } } add_trait = honest }
	if = { limit = { NOR = { has_trait = brave has_trait = craven } } add_trait = brave }
	if = { limit = { NOR = { has_trait = gregarious has_trait = shy } } add_trait = gregarious }
	if = { limit = { NOR = { has_trait = content has_trait = ambitious } } add_trait = content }
	if = { limit = { NOR = { has_trait = just has_trait = arbitrary } } add_trait = just }
	if = { limit = { NOR = { has_trait = zealous has_trait = cynical } } add_trait = zealous }
	if = { limit = { NOR = { has_trait = paranoid has_trait = trusting } } add_trait = paranoid }
	if = {
		limit = {
			NOR = {
				has_trait = compassionate
				has_trait = callous
				has_trait = sadistic
			}
		}
		add_trait = compassionate
	}
	if = { limit = { NOR = { has_trait = forgiving has_trait = vengeful } } add_trait = forgiving }
}

# Replaces one vanilla opposite before adding the requested personality trait.
DICM_RBE_add_personality_trait_with_one_opposite_effect = {
	if = { limit = { has_trait = $OPPOSITE$ } remove_trait = $OPPOSITE$ }
	if = { limit = { NOT = { has_trait = $TRAIT$ } } add_trait = $TRAIT$ }
}

# Replaces two vanilla opposites before adding the requested personality trait.
DICM_RBE_add_personality_trait_with_two_opposites_effect = {

	if = { limit = { NOT = { has_trait = $TRAIT$ } } add_trait = $TRAIT$ }
}

# Applies the selected Caesar-family personality setting to adults only.
DICM_RBE_apply_ceasar_family_personality_setting_effect = {
	if = {
		limit = {
			is_adult = yes
		}
		if = {
			limit = {
				DICM_RBE_pious_dynasty_enabled_trigger = yes
			}
			DICM_RBE_add_chaste_trait_effect = yes
		}
		else_if = {
			limit = {
				DICM_RBE_never_too_many_bastards_enabled_trigger = yes
			}
			DICM_RBE_add_lustful_trait_effect = yes
		}
	}
}
